    function openAssetDatabase() {
      return new Promise((resolve, reject) => {
        if (!("indexedDB" in window)) {
          reject(new Error("IndexedDB no está disponible en este navegador."));
          return;
        }
        const request = indexedDB.open(ASSET_DB_NAME, ASSET_DB_VERSION);
        request.onupgradeneeded = () => {
          const db = request.result;
          if (!db.objectStoreNames.contains(ASSET_STORE_NAME)) {
            db.createObjectStore(ASSET_STORE_NAME, { keyPath: "name" });
          }
          if (!db.objectStoreNames.contains(PROJECT_STORE_NAME)) {
            db.createObjectStore(PROJECT_STORE_NAME, { keyPath: "id" });
          }
        };
        request.onsuccess = () => resolve(request.result);
        request.onerror = () => reject(request.error || new Error("No se pudo abrir IndexedDB."));
      });
    }

    function assetDbRequest(mode, action) {
      return openAssetDatabase().then(db => new Promise((resolve, reject) => {
        const tx = db.transaction(ASSET_STORE_NAME, mode);
        const store = tx.objectStore(ASSET_STORE_NAME);
        let request;
        try {
          request = action(store);
        } catch (error) {
          db.close();
          reject(error);
          return;
        }
        request.onsuccess = () => resolve(request.result);
        request.onerror = () => reject(request.error || new Error("Error en IndexedDB."));
        tx.oncomplete = () => db.close();
        tx.onerror = () => {
          db.close();
          reject(tx.error || new Error("Error guardando assets."));
        };
      }));
    }

    function persistAsset(name, bytes) {
      const copy = bytes instanceof Uint8Array ? bytes.slice().buffer : bytes;
      return assetDbRequest("readwrite", store =>
        store.put({ name, bytes: copy, updatedAt: Date.now() })
      );
    }

    function assetMimeType(name) {
      const ext = name.toLowerCase().split(".").pop();
      const types = {
        png:"image/png", jpg:"image/jpeg", jpeg:"image/jpeg", gif:"image/gif",
        webp:"image/webp", avif:"image/avif", svg:"image/svg+xml",
        mp4:"video/mp4", webm:"video/webm", mov:"video/quicktime",
        ogv:"video/ogg", mp3:"audio/mpeg", wav:"audio/wav", ogg:"audio/ogg",
        m4a:"audio/mp4", json:"application/json", txt:"text/plain",
        csv:"text/csv", ttf:"font/ttf", otf:"font/otf", woff:"font/woff",
        woff2:"font/woff2",
      };
      return types[ext] || "application/octet-stream";
    }

    function assetUrl(path) {
      const oldUrl = assetObjectUrls.get(path);
      if (oldUrl) URL.revokeObjectURL(oldUrl);
      const bytes = pyodide.FS.readFile(path);
      const url = URL.createObjectURL(new Blob([bytes], {type: assetMimeType(path)}));
      assetObjectUrls.set(path, url);
      return url;
    }

      }
    }

    function formatBytes(bytes) {
      if (bytes < 1024) return bytes + " B";
      if (bytes < 1024 ** 2) return (bytes / 1024).toFixed(1) + " KB";
      return (bytes / 1024 ** 2).toFixed(1) + " MB";
    }

    function uniqueAssetPath(fileName) {
      const safeName = fileName.replace(/[/\\]/g, "_").trim() || "asset";
      const dot = safeName.lastIndexOf(".");
      const stem = dot > 0 ? safeName.slice(0, dot) : safeName;
      const ext = dot > 0 ? safeName.slice(dot) : "";
      let candidate = ASSET_DIR + "/" + safeName;
      let index = 1;
      while (pyodide.FS.analyzePath(candidate).exists) {
        candidate = ASSET_DIR + "/" + stem + "_" + index++ + ext;
      }
      return candidate;
    }

    async function importAssets(files) {
      if (!pyodide || !files?.length) return;
      ensureAssetDir();

      let imported = 0;
      for (const file of Array.from(files)) {
        if (!file || typeof file.arrayBuffer !== "function") continue;
        const bytes = new Uint8Array(await file.arrayBuffer());
        const path = uniqueAssetPath(file.name);
        const storedName = path.slice(ASSET_DIR.length + 1);
        pyodide.FS.writeFile(path, bytes);
        try {
          await persistAsset(storedName, bytes);
        } catch (error) {
          console.warn("[asset persistence] No se pudo guardar el asset:", storedName, error);
          setStatus("Asset importado, pero no se pudo guardar de forma persistente: " + storedName);
        }

        const extension = file.name.toLowerCase().split(".").pop();
        if (["ttf", "otf", "woff", "woff2"].includes(extension)) {
          const metadata = await readFontMetadata(path);
          await registerFontAsset(path);
          assetFontMetadata.set(path, metadata);
        }

        imported++;
      }

      refreshAssets();
      if (imported) {
        setStatus(imported === 1
          ? "Asset importado."
          : imported + " assets importados.");
      }
    }

    async function importClipboardData(clipboardData) {
      if (!clipboardData) return;

      const files = Array.from(clipboardData.files || []);
      if (files.length) {
        await importAssets(files);
        return true;
      }

      const items = Array.from(clipboardData.items || []);
      const imageItems = items.filter(item =>
        item.kind === "file" && item.type.startsWith("image/")
      );

      if (!imageItems.length) return false;

      const filesFromClipboard = [];
      for (const item of imageItems) {
        const blob = item.getAsFile();
        if (!blob) continue;
        const extension = blob.type.split("/")[1]?.replace("jpeg", "jpg") || "png";
        filesFromClipboard.push(
          new File([blob], "pasted-image." + extension, {type: blob.type})
        );
      }

      await importAssets(filesFromClipboard);
      return filesFromClipboard.length > 0;
    }

          }
          micStream = await navigator.mediaDevices.getUserMedia({ audio: true });
          recordingMicSource = context.createMediaStreamSource(micStream);
          recordingMicSource.connect(destination);
        }

        const requestedFps = Number(recordFpsSelect.value) || 60;
        stream = canvas.captureStream(requestedFps);
        for (const track of destination.stream.getAudioTracks()) stream.addTrack(track);
        const mimeType = getRecordingMimeType();
        mediaRecorder = mimeType
          ? new MediaRecorder(stream, { mimeType })
          : new MediaRecorder(stream);
        recordingChunks = [];
        recordingStream = stream;
        recordingMicStream = micStream;
        recordingAudioDestination = destination;

        mediaRecorder.addEventListener("dataavailable", event => {
          if (event.data && event.data.size > 0) recordingChunks.push(event.data);
        });
        mediaRecorder.addEventListener("error", event => {
          const error = event.error || new Error("Error durante la grabación.");
          showError(error);
          setStatus("Error al grabar la escena.");
        });
        mediaRecorder.addEventListener("stop", () => {
          const type = mediaRecorder?.mimeType || "video/webm";
          const blob = new Blob(recordingChunks, { type });
          if (blob.size > 0) {
            const url = URL.createObjectURL(blob);
            const link = document.createElement("a");
            link.href = url;
            link.download = "manimgl-" + new Date().toISOString().replace(/[:.]/g, "-") + ".webm";
            document.body.appendChild(link);
            link.click();
            link.remove();
            setTimeout(() => URL.revokeObjectURL(url), 60_000);
            setStatus("Grabación lista: se descargó el vídeo con audio.");
          } else {
            setStatus("La grabación terminó, pero no se generaron datos de vídeo.");
          }
          for (const track of recordingStream?.getTracks() || []) track.stop();
          for (const track of recordingMicStream?.getTracks() || []) track.stop();
          try {
            if (recordingMicSource && recordingAudioDestination) {
              recordingMicSource.disconnect(recordingAudioDestination);
            }
          } catch (_) {}
          if (recordingAudioDestination) {
            window.__manimDisconnectRecordingAudio(recordingAudioDestination);
          }
          recordingStream = null;
          recordingMicStream = null;
          recordingMicSource = null;
          recordingAudioDestination = null;
          mediaRecorder = null;
          recordingChunks = [];
          recordButton.disabled = false;
          recordButton.classList.remove("recording");
          recordButton.textContent = "● Grabar";
          recordStopButton.disabled = true;
          runRecordButton.disabled = !pyodide;
        });
        mediaRecorder.start(1000);
        recordButton.disabled = true;
        recordButton.classList.add("recording");
        recordButton.textContent = "● Grabando…";
        recordStopButton.disabled = false;
        runRecordButton.disabled = true;
        setStatus(recordMicCheckbox.checked
          ? "Grabando escena y micrófono…"
          : "Grabando escena y audio…");
      } catch (error) {
        if (stream) for (const track of stream.getTracks()) track.stop();
        if (micStream) for (const track of micStream.getTracks()) track.stop();
        if (destination) window.__manimDisconnectRecordingAudio(destination);
        if (recordingMicSource && destination) {
          try { recordingMicSource.disconnect(destination); } catch (_) {}
        }
        recordingMicSource = null;
        throw error;
      }
    }

        async function registerCatalogFont(fileName, url) {
          const response = await fetch(url, { cache: "force-cache" });
          if (!response.ok) {
            throw new Error("HTTP " + response.status + " al cargar " + fileName);
          }

          const bytes = await response.arrayBuffer();
          if (bytes.byteLength < 4) throw new Error("Archivo de fuente vacío o incompleto");
          const signature = new Uint8Array(bytes, 0, 4);
          const signatureText = Array.from(signature, byte =>
            byte.toString(16).padStart(2, "0")
          ).join("");
          const extension = fileName.toLowerCase().split(".").pop();
          if (extension === "ttf" &&
              !(signatureText.startsWith("00010000") ||
                signatureText.startsWith("4f54544f") ||
                signatureText.startsWith("74727565") ||
                signatureText.startsWith("74797031") ||
                signatureText.startsWith("74746366"))) {
            throw new Error("El archivo .ttf no tiene una firma TrueType válida (firma: " + signatureText + ")");
          }

          const result = { name: fileName, pango: null, typst: false, errors: [] };
          try {
            result.pango = await window.manimRegisterPangoFont(bytes.slice(0), fileName);
          } catch (error) {
            result.errors.push("Pango: " + String(error?.message || error));
          }
          try {
            const compiler = window.manimTypstCompiler;
            if (!compiler || typeof compiler.addFonts !== "function") {
              throw new Error("El compilador Typst del navegador no expone addFonts()");
            }
            await compiler.addFonts(...[new Uint8Array(bytes.slice(0))]);
            result.typst = true;
          } catch (error) {
            result.errors.push("Typst: " + String(error?.message || error));
          }
          if (!result.pango && !result.typst) {
            throw new Error(result.errors.join(" | ") || "Ningún motor aceptó la fuente");
          }
          return result;
        }

        const fontManifestUrl = new URL("./fonts/fonts.json", document.baseURI);
        const fontManifestResponse = await fetch(fontManifestUrl, { cache: "force-cache" });
        if (!fontManifestResponse.ok) {
          throw new Error("No se pudo cargar fonts/fonts.json: HTTP " + fontManifestResponse.status);
        }
        const fontManifest = await fontManifestResponse.json();
        if (!Array.isArray(fontManifest)) {
          throw new Error("fonts/fonts.json debe contener un array de fuentes.");
        }
        const availableFonts = fontManifest.filter(font =>

        async function loadCatalogFont(font) {
          const fileName = font.fileName || font.name;
          if (loadedFontNames.has(fileName) || loadingFontNames.has(fileName)) return;
          loadingFontNames.add(fileName);
          failedFontNames.delete(fileName);
          renderFontCatalog();
          updateFontsSummary("Cargando " + fileName + "…");
          try {
            const url = new URL(font.url, fontManifestUrl).href;
            const result = await registerCatalogFont(fileName, url);
            if (result.pango) {
              fontStats.pango++;
              window.manimBrowserFontFamily = String(result.pango.family);
            }
            if (result.typst) fontStats.typst++;
            loadedFontNames.add(fileName);
            selectedFontNames.delete(fileName);
            fontStats.success++;
            if (result.errors.length) {
              failedFontNames.set(fileName, result.errors.join(" | "));
              console.warn("[font catalog] Registro parcial:", fileName, result.errors);
            }
          } catch (error) {
            failedFontNames.set(fileName, String(error?.message || error));
            console.warn("[font catalog]", fileName, error);
          } finally {
            loadingFontNames.delete(fileName);
            renderFontCatalog();
          }
        }

        async function loadFontBatch(fonts) {
          const pending = fonts.filter(font => {
            const fileName = font.fileName || font.name;
            return !loadedFontNames.has(fileName) && !loadingFontNames.has(fileName);
          });
          if (!pending.length) {
            updateFontsSummary("Las fuentes seleccionadas ya están cargadas.");
            return;
          }
          fontsLoadSelected.disabled = true;
          fontsLoadAll.disabled = true;
          let done = 0;
          try {
            for (const font of pending) {
              await loadCatalogFont(font);
              done++;
              updateFontsSummary("Cargando fuentes adicionales: " + done + "/" + pending.length);
            }
          } finally {
            fontsLoadSelected.disabled = false;
            fontsLoadAll.disabled = false;
            renderFontCatalog();
            updateFontsSummary();
          }
        }
